sts_liang_present_daughter = {
	decision_category = country_specific_decisions
	tags = { KFE }
	# 成功标记保证仅完成一次；失败不消费引擎 only_once 次数。

	potential = {
		tag = KFE
		NOT = { has_variable = sts_liang_empress_installed }
	}
	allow = {
		sts_can_present_daughter = yes
	}
	# 原版最高 AI 意愿值（格拉纳达决议同用 1000），条件满足时 AI 开局即优先执行。
	ai_will_do = { value = 1000 }

	option = {
		name = sts_liang_present_daughter.a
		ai_chance = { value = 100 }
		effect = {
			if = {
				limit = { sts_can_present_daughter = yes }
				character:sts_kfe_daughter1 = { move_country = c:CSO }
				c:CSO.ruler = {
					marry_character = character:sts_kfe_daughter1
				}
				# 婚姻实际产生宋国配偶后，才确认册后并广播；失败不领任何奖励。
				if = {
					limit = {
						c:CSO = {
							has_consort = yes
							consort = character:sts_kfe_daughter1
						}
					}
					set_variable = { name = sts_liang_empress_installed value = yes }
					trigger_event_non_silently = sts_liang_empress.1
					every_country = {
						limit = {
							country_type = location
							sts_is_consolidation_member = yes
							NOT = { tag = KFE }
							NOT = { has_variable = sts_liang_empress_news_seen }
						}
						set_variable = { name = sts_liang_empress_news_seen value = yes }
						trigger_event_non_silently = sts_liang_empress.2
					}
				}
				else = {
					# 若引擎没有建立预期配偶身份，送回梁，不播报、不发奖。
					character:sts_kfe_daughter1 = { move_country = c:KFE }
				}
			}
		}
	}
}
